+// PROBE ONLY (replay 5b feasibility, before the plan request): the agreed
+// HTTP instrumentation of FI265-02 Rest A (R268/R269) in its smallest form —
+// a catch-all route registered FIRST on every context (the lowest priority)
+// that continues each request no spec handler took with `x-e2e-req: <id>`,
+// and the browser's own request events with that id, written to a file of
+// this worker beside the event log. Measures: does an always-on catch-all
+// change what the tests see (counts, unload beacons, run time); are worker
+// requests reported and routed; does the header change a request; does the
+// route handler see the request after its `request` event.
+import { appendFileSync } from "node:fs";
+import type { BrowserContext, Request } from "@playwright/test";
+
+const OUT = process.env.LCNC_E2E_EVENTS ? `${process.env.LCNC_E2E_EVENTS}.http-${process.env.TEST_WORKER_INDEX ?? "x"}.jsonl` : null;
+const write = (rec: Record<string, unknown>) => { if (OUT) appendFileSync(OUT, JSON.stringify({ t: Date.now(), ...rec }) + "\n"); };
+
+let seq = 0;
+
+const done = new WeakSet<BrowserContext>();
+
+export async function instrument(context: BrowserContext, origin: { test: string; retry: number }, label: string): Promise<void> {
+  if (done.has(context)) return;
+  done.add(context);
+  const ids = new WeakMap<Request, string>();
+  const idOf = (r: Request): string => {
+    let id = ids.get(r);
+    if (!id) { id = `${origin.test}.${origin.retry}.${++seq}`; ids.set(r, id); }
+    return id;
+  };
+  context.on("request", (r) => {
+    const u = new URL(r.url());
+    const from = r.redirectedFrom();
+    write({ kind: "breq", ctx: label, test: origin.test, retry: origin.retry, id: idOf(r), method: r.method(), path: u.pathname,
+            qkeys: [...u.searchParams.keys()], type: r.resourceType(), sw: r.serviceWorker() !== null,
+            from: from ? idOf(from) : null, host: u.host });
+  });
+  context.on("requestfinished", async (r) => {
+    const resp = await r.response().catch(() => null);
+    write({ kind: "bfin", id: idOf(r), status: resp?.status() ?? null });
+  });
+  context.on("requestfailed", (r) => write({ kind: "bfail", id: idOf(r), error: r.failure()?.errorText ?? null }));
+  await context.route(() => true, async (route) => {
+    const r = route.request();
+    const known = ids.has(r);
+    const id = idOf(r);
+    write({ kind: "broute", id, known_at_route: known });
+    await route.continue({ headers: { ...r.headers(), "x-e2e-req": id } });
+  });
+}
