49:         """Full tool_library.json (all INI configs), mtime-cached. Hot path:
50:         the status loop calls load() on every tool-number change per client.
51:         Keyed by (inode, mtime_ns, size): catches UI edits and external edits.
52:         `strict`: a failed stat, read or parse, or a top level that is not an
53:         object, raises ToolLibraryUnreadable instead of answering the last
54:         cached data or {} — from the cache too (Codex R468 MT-I36): the
55:         tolerant status reader shares it and caches what it read, whatever
56:         its form, so a hit is no proof of the form."""
57:         if not self._path.exists():
58:             self._cache = None
59:             return {}
60:         try:
61:             st = self._path.stat()
62:             sig = (st.st_ino, st.st_mtime_ns, st.st_size)
63:         except OSError as e:
64:             self._emit("tool_lib.stat_failed", "warn", e)
65:             if strict:
66:                 raise ToolLibraryUnreadable(f"not read ({e.strerror})") from None
67:             return self._cache[1] if self._cache else {}
68:         if self._cache is not None and self._cache[0] == sig:
69:             if strict:
70:                 self._require_object(self._cache[1])
71:             return self._cache[1]
72:         try:
73:             with open(self._path, "r") as f:
74:                 data = json.load(f)
75:         except json.JSONDecodeError as e:
76:             self._emit("tool_lib.corrupt", "error", e)
77:             if strict:
78:                 raise ToolLibraryUnreadable(f"not JSON (line {e.lineno})") from None
79:             return self._cache[1] if self._cache else {}
80:         except OSError as e:
81:             self._emit("tool_lib.read_failed", "warn", e)
82:             if strict:
83:                 raise ToolLibraryUnreadable(f"not read ({e.strerror})") from None
84:             return self._cache[1] if self._cache else {}
85:         if strict:
86:             self._require_object(data)
87:         self._cache = (sig, data)
88:         return data

99:     def load(self, strict: bool = False, *, ini_key: Optional[str] = None) -> dict:
100:         """Tool metadata for the current INI config — or for `ini_key`: what a
101:         read or an edit was bound to at its start, never another one's.
102: 
103:         Migrates the old flat format (top-level keys that look like tool numbers)
104:         by wrapping it under the current INI key once. Lock-guarded so a
105:         concurrent save() (different executor thread) can't tear the cache.
106:         `strict` (the Tools tab's read, P6 "tool files read"): an unreadable
107:         file, a section or a seed that is not an object raise
108:         ToolLibraryUnreadable — never the cached rows or {} as the library."""
109:         with self._lock:
110:             all_data = self._load_all(strict)
111:             ini = self._ini_key() if ini_key is None else ini_key
112:             # (an object only: a list of digit strings is no flat library — the
113:             # tolerant reader rewrote it, and the strict answer followed, MT-I36)
114:             if (isinstance(all_data, dict) and all_data and not any(k.startswith("/") for k in all_data)
115:                     and any(k.isdigit() for k in all_data)):
116:                 all_data = {ini: all_data}
117:                 self._save_all(all_data)
118:             if ini in all_data:
119:                 if strict and not isinstance(all_data[ini], dict):
120:                     raise ToolLibraryUnreadable(f"section is {type(all_data[ini]).__name__}, not an object")
121:                 return all_data[ini]  # An explicitly saved empty table stays empty.
122:             # Fresh example installations carry geometry beside their tool.tbl.
123:             # Never overlay a saved library, including legacy flat libraries.
124:             path = self._seed_path() if self._seed_path else None
125:             if path is None or not path.is_file():
126:                 return {}
127:             try:
128:                 key = (str(path), path.stat().st_mtime_ns)
129:                 if self._seed_cache is None or self._seed_cache[0] != key:
130:                     data = json.loads(path.read_text())
131:                     if not isinstance(data, dict):
132:                         raise ValueError('Tool metadata seed must be an object')
133:                     self._seed_cache = (key, data)
134:                 return self._seed_cache[1]
135:             except (OSError, ValueError) as e:
136:                 self._emit('tool_lib.seed_failed', 'warn', e)
137:                 if strict:
138:                     raise ToolLibraryUnreadable(f"seed not read ({e})") from None
139:                 return {}
