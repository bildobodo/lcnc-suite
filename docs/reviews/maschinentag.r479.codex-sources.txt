288:     """{table index: (T, P)} of the home entries LinuxCNC holds after it

295:         if free == _INDEX_END:
296:             break                                   # every index is taken: no later line holds one
297:         if b"\0" in piece:
298:             raise ToolTableUnreadable(f"Line {n} not readable — fix it in an editor")
299:         if piece.startswith(b";"):
300:             continue
301:         part = piece.lstrip(b";").split(b";", 1)[0]
302:         if len(part) <= 1:
303:             continue
304:         tool, index, pocket, valid = -1, 0, 0, True
305:         for token in part.split(b" "):
306:             if not token:
307:                 continue
308:             letter, rest = token[:1].upper(), token[1:]
309:             if letter in (b"T", b"P", b"Q"):
310:                 v = _c_int(rest)
311:                 if v is None:
312:                     valid = False
313:                 elif letter == b"T":
314:                     tool = v
315:                 elif letter == b"P":
316:                     pocket = v
317:                     if free < _INDEX_END:
318:                         index, free = free, free + 1
319:                     else:                           # "Out-of-range nonrandom_idx"
320:                         valid = False
321:             elif letter in b"DXYZABCUVWIJ":
322:                 if _C_REAL_OTHER_RE.match(rest):
323:                     raise ToolTableUnreadable(f"Line {n} not readable — fix it in an editor")
324:                 if _C_REAL_RE.match(rest) is None:
325:                     valid = False
326:             elif not token.startswith(b"\n"):
327:                 valid = False
328:         if valid and index > 0 and tool != -1:      # (index 0 is the spindle's; T-1 is LinuxCNC's "no tool")
329:             out[index] = (tool, pocket)
330:     return out
